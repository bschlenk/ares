add_library(ruby STATIC ruby.cpp)
add_library(ares::ruby ALIAS ruby)

find_package(SDL REQUIRED)
target_sources(ruby PRIVATE ruby.hpp audio/audio.cpp audio/audio.hpp)
set_source_files_properties(audio/audio.cpp PROPERTIES HEADER_FILE_ONLY TRUE)
target_include_directories(ruby PRIVATE ${CMAKE_SOURCE_DIR})
target_link_libraries(ruby PRIVATE ares::nall::headers)
if(OS_MACOS)
  target_link_libraries(ruby PRIVATE "$<LINK_LIBRARY:FRAMEWORK,SDL::SDL>")
else()
  target_link_libraries(ruby PRIVATE SDL::SDL)
endif()
if(OS_WINDOWS)
  target_link_libraries(ruby PRIVATE winmm)
endif()
target_enable_feature(ruby "SDL audio driver" AUDIO_SDL)
